CREATE TABLE "Sist_Roles" (
    "Id_Rol" INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    "Nombre" TEXT NOT NULL,
    "Descripcion" TEXT,
    "Activo" BOOLEAN DEFAULT TRUE,
    "Es_Predeterminado" BOOLEAN DEFAULT FALSE,
    "Icono" TEXT DEFAULT 'fa-user-shield',
    "Fecha_Creacion" TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE "Sist_Modulos" (
    "Id_Modulo" INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    "Nombre_Clave" TEXT NOT NULL,

    "Activo" BOOLEAN DEFAULT TRUE,
    "Fecha_Creacion" TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE "Sist_Usuarios" (
    "Id_Usuario" INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    "NombreCompleto" TEXT NOT NULL,
    "Nombre_Usuario" TEXT UNIQUE,
    "Email" TEXT NOT NULL UNIQUE,

    "Telefono" TEXT,

    "Id_Permiso" INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

CREATE TABLE "Fin_Caja" (
    "Id_Movimiento" SERIAL PRIMARY KEY,
    "Concepto" TEXT NOT NULL,
    "Monto" DECIMAL(12,2) NOT NULL,
    "Tipo" TEXT CHECK ("Tipo" IN ('Ingreso', 'Egreso')),
    "Fecha" TIMESTAMP DEFAULT NOW(),
    "Mes" INT GENERATED ALWAYS AS (EXTRACT(MONTH FROM "Fecha")) STORED,
    "Anio" INT GENERATED ALWAYS AS (EXTRACT(YEAR FROM "Fecha")) STORED,
    "Id_Usuario" INT REFERENCES "Sist_Usuarios"("Id_Usuario"),
    "Id_Archivo" INT REFERENCES "Rec_Archivos"("Id_Archivo"),
    "Movimiento_En_Banco" BOOLEAN DEFAULT FALSE,
    "Editado" BOOLEAN NOT NULL DEFAULT FALSE,
    "Folio_Bancario" VARCHAR(100),
    "Id_Evento" INTEGER NULL
);

CREATE TABLE "Fin_Caja_Historial" (
    "Id_Historial" SERIAL PRIMARY KEY,
    "Id_Movimiento" INTEGER NOT NULL REFERENCES "Fin_Caja"("Id_Movimiento") ON DELETE CASCADE,
    "Monto_Anterior" DECIMAL(18,2) NOT NULL,
    "Monto_Nuevo" DECIMAL(18,2) NOT NULL,
    "Concepto_Anterior" VARCHAR(255),
    "Concepto_Nuevo" VARCHAR(255),
    "Tipo_Anterior" TEXT,
    "Tipo_Nuevo" TEXT,
    "Fecha_Edicion" TIMESTAMP NOT NULL DEFAULT NOW(),
    "Id_Usuario_Editor" INTEGER NOT NULL REFERENCES "Sist_Usuarios"("Id_Usuario")
);

    "Monto" DECIMAL(18,2) NOT NULL,

    "Concepto" VARCHAR(255) NOT NULL,
    "Fecha_Solicitud" TIMESTAMP NOT NULL DEFAULT NOW(),
    "Fecha_Resolucion" TIMESTAMP NULL
);

CREATE TABLE "Fin_Proyectos" (

-- ============================================================================
-- SCRIPT DE MIGRACIÓN: COLUMNA Id_Asistente_Evento EN ENCUESTAS
-- Motor: PostgreSQL
-- Tabla afectada: "Encuestas_Respuestas_Header"
-- ============================================================================

-- 1. Agregar la columna Id_Asistente_Evento
ALTER TABLE "Encuestas_Respuestas_Header" 
ADD COLUMN IF NOT EXISTS "Id_Asistente_Evento" INTEGER NULL;

-- 2. Clave foránea opcional hacia Eventos_B_Asistentes
DO $$
BEGIN
    IF NOT EXISTS (
        SELECT 1 FROM pg_constraint WHERE conname = 'FK_Encuestas_Respuestas_Asistente_Evento'
    ) THEN
        ALTER TABLE "Encuestas_Respuestas_Header"
        ADD CONSTRAINT "FK_Encuestas_Respuestas_Asistente_Evento"
        FOREIGN KEY ("Id_Asistente_Evento") 
        REFERENCES "Eventos_B_Asistentes"("Id_Asistente")
        ON DELETE SET NULL;
    END IF;
END $$;

-- 3. Índice para optimizar la verificación instantánea de encuestas
CREATE INDEX IF NOT EXISTS "IX_Encuestas_Respuestas_Asistente_Evento" 
ON "Encuestas_Respuestas_Header" ("Id_Encuesta", "Id_Asistente_Evento");
